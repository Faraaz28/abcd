G Continue And Break
public class ContinueAndBreak {
    public static void main(String[] args) {
        for (int i = 1; i <= 5; i++) {
            if (i == 3) {
                continue; // Skip iteration when i is 3
            }

            if (i == 4) {
                break; // Exit the loop when i is 4
            }

            System.out.println("Iteration " + i);
        }
    }
}

Constructor chaining
class Vehicle {
    String type;

    // Constructor of Vehicle class
    Vehicle(String t) {
        type = t;
    }

    void display() {
        System.out.println("Vehicle Type: " + type);
    }
}

class Car extends Vehicle {
    String model;
    int year;

    // Constructor chaining to call superclass constructor
    Car(String m, int y, String t) {
        super(t);  // Calling the Vehicle class constructor
        model = m;
        year = y;
    }

    void display() {
        super.display();  // Calling display method of superclass
        System.out.println("Model: " + model);
        System.out.println("Year: " + year);
    }
}

public class ConstructorChainingExample {
    public static void main(String[] args) {
        // Creating an object with constructor chaining
        Car car = new Car("Tesla Model S", 2023, "Electric");
        
        // Displaying car details
        car.display();
    }
}

5 Polymorphism
class Animal {
    public void sound() {
        System.out.println("This is an animal sound.");
    }
}

class Dog extends Animal {
    @Override
    public void sound() {
        System.out.println("Dog barks.");
    }
}

class Cat extends Animal {
    @Override
    public void sound() {
        System.out.println("Cat meows.");
    }
}

public class PolymorphismExample {
    public static void main(String[] args) {
        Animal a1 = new Dog(); // Dog instance, but referred to by Animal reference
        Animal a2 = new Cat(); // Cat instance, but referred to by Animal reference

        a1.sound(); // Output: Dog barks.
        a2.sound(); // Output: Cat meows.
    }
}

Method Overloading
class Calculator {
    // Overloaded method with two integers
    public int add(int a, int b) {
        return a + b;
    }

    // Overloaded method with three integers
    public int add(int a, int b, int c) {
        return a + b + c;
    }

    // Overloaded method with two double parameters
    public double add(double a, double b) {
        return a + b;
    }
}

public class MethodOverloadingExample {
    public static void main(String[] args) {
        Calculator calc = new Calculator();
        System.out.println("Sum of two integers: " + calc.add(10, 20)); // Output: 30
        System.out.println("Sum of three integers: " + calc.add(10, 20, 30)); // Output: 60
        System.out.println("Sum of two doubles: " + calc.add(10.5, 20.5)); // Output: 31.0
    }
}

Method Overriding 
class Parent {
    public void show() {
        System.out.println("Parent's show method.");
    }
}

class Child extends Parent {
    @Override
    public void show() {
        System.out.println("Child's show method.");
    }
}

public class MethodOverridingExample {
    public static void main(String[] args) {
        Parent p = new Parent();
        Child c = new Child();
        Parent pc = new Child();

        p.show();   // Output: Parent's show method.
        c.show();   // Output: Child's show method.
        pc.show();  // Output: Child's show method.
    }
}



Constructor overloading
class Student {
    String name;
    int age;
    String course;

    // Constructor with no parameters
    public Student() {
        this.name = "Unknown";
        this.age = 0;
        this.course = "Unknown";
    }

    // Constructor with one parameter
    public Student(String name) {
        this.name = name;
        this.age = 0;
        this.course = "Unknown";
    }

    // Constructor with two parameters
    public Student(String name, int age) {
        this.name = name;
        this.age = age;
        this.course = "Unknown";
    }

    // Constructor with three parameters
    public Student(String name, int age, String course) {
        this.name = name;
        this.age = age;
        this.course = course;
    }

    public void displayInfo() {
        System.out.println("Name: " + name + ", Age: " + age + ", Course: " + course);
    }
}

public class ConstructorOverloadingExample {
    public static void main(String[] args) {
        Student s1 = new Student();
        Student s2 = new Student("John");
        Student s3 = new Student("John", 20);
        Student s4 = new Student("John", 20, "Computer Science");

        s1.displayInfo(); // Output: Name: Unknown, Age: 0, Course: Unknown
        s2.displayInfo(); // Output: Name: John, Age: 0, Course: Unknown
        s3.displayInfo(); // Output: Name: John, Age: 20, Course: Unknown
        s4.displayInfo(); // Output: Name: John, Age: 20, Course: Computer Science
    }
}
